const API_URL = window.location.origin; let ticketActual = null; // Crear Ticket document.getElementById('formulario-ticket').addEventListener('submit', async (e) => { e.preventDefault(); const email = document.getElementById('email').value; const nombre = document.getElementById('nombre').value; const asunto = document.getElementById('asunto').value; const descripcion = document.getElementById('descripcion').value; const prioridad = document.getElementById('prioridad').value; try { const response = await fetch(`${API_URL}/api/tickets`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, nombre, asunto, descripcion, prioridad }) }); const data = await response.json(); if (data.id) { const mensajeDiv = document.getElementById('mensaje-exito'); mensajeDiv.textContent = `✅ Ticket creado exitosamente. ID: ${data.id}. Guarda este ID para seguimiento.`; mensajeDiv.style.display = 'block'; document.getElementById('formulario-ticket').reset(); } } catch (error) { console.error('Error:', error); alert('Error al crear ticket'); } }); // Buscar Tickets function buscarTickets() { const email = document.getElementById('email-buscar').value; if (!email) { alert('Ingresa tu email'); return; } fetch(`${API_URL}/api/tickets-usuario/${encodeURIComponent(email)}`) .then(res => res.json()) .then(tickets => { const lista = document.getElementById('lista-tickets'); lista.innerHTML = ''; if (tickets.length === 0) { lista.innerHTML = '

No hay tickets para este email

'; return; } tickets.forEach(ticket => { const div = document.createElement('div'); div.className = 'card mb-2'; div.innerHTML = `
${ticket.asunto}
Estado: ${ticket.estado}
`; lista.appendChild(div); }); }) .catch(err => { console.error('Error:', err); alert('Error al buscar tickets'); }); } // Ver Detalles del Ticket function verTicket(ticketId) { ticketActual = ticketId; fetch(`${API_URL}/api/tickets/${ticketId}`) .then(res => res.json()) .then(ticket => { document.getElementById('titulo-ticket').textContent = `Ticket #${ticket.id}`; document.getElementById('contenido-ticket').innerHTML = `

Nombre: ${ticket.nombre}

Email: ${ticket.email}

Asunto: ${ticket.asunto}

Descripción: ${ticket.descripcion}

Estado: ${ticket.estado}

Prioridad: ${ticket.prioridad}

`; const listaMensajes = document.getElementById('lista-mensajes'); listaMensajes.innerHTML = ''; if (ticket.mensajes && ticket.mensajes.length > 0) { ticket.mensajes.forEach(msg => { const p = document.createElement('div'); p.className = 'mb-2 p-2 border-bottom'; p.innerHTML = `${msg.autor}: ${msg.mensaje}
${new Date(msg.fecha).toLocaleString()}`; listaMensajes.appendChild(p); }); } else { listaMensajes.innerHTML = '

Sin mensajes aún

'; } mostrarSeccion('ticket'); }) .catch(err => { console.error('Error:', err); alert('Error al cargar ticket'); }); } // Agregar Mensaje function agregarMensaje() { const mensaje = document.getElementById('nuevo-mensaje').value; if (!mensaje) { alert('Escribe un mensaje'); return; } fetch(`${API_URL}/api/mensajes`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ticket_id: ticketActual, autor: 'Usuario', tipo_autor: 'usuario', mensaje: mensaje }) }) .then(res => res.json()) .then(data => { document.getElementById('nuevo-mensaje').value = ''; verTicket(ticketActual); }) .catch(err => { console.error('Error:', err); alert('Error al enviar mensaje'); }); } // Cambiar Sección function mostrarSeccion(seccion) { document.querySelectorAll('.seccion').forEach(s => s.classList.remove('active')); document.getElementById('seccion-' + seccion).classList.add('active'); }
